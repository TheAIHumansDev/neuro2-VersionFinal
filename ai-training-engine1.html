// Se ejecuta cuando el usuario habla con el MetaHuman
async function queryNeuroCore(userSpeech, sessionId) {
  const res = await fetch("https://api.neuroai.app/v1/metahuman/query", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${NEURO_API_KEY}`,
      "Content-Type": "application/json"
    },
    body: JSON.stringify({
      query: userSpeech,
      session_id: sessionId,
      persona_id: "sofia-concierge-v2",
      mode: "rag_only",
      // el backend recupera únicamente fragmentos reales de tus documentos indexados
      stream: true
    })
  });

  const data = await res.json();

  // data.response.text -> se envía al motor de voz (TTS)
  // data.response.avatar.emotion / gesture -> controla la animación facial
  avatarEngine.speak(data.response.text, {
    emotion: data.response.avatar.emotion,
    gesture: data.response.avatar.gesture
  });

  return data;
}

// Sincroniza un Google Sheet como fuente de conocimiento del MetaHuman
function syncSheetToNeuroCore() {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("FAQ");
  const rows = sheet.getDataRange().getValues().slice(1);

  const payload = {
    source_id: "google_sheets_faq",
    source_type: "structured",
    records: rows.map(r => ({ pregunta: r[0], respuesta: r[1], categoria: r[2] }))
  };

  UrlFetchApp.fetch("https://api.neuroai.app/v1/sources/sync", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${PropertiesService.getScriptProperties().getProperty("NEURO_API_KEY")}`,
      "Content-Type": "application/json"
    },
    payload: JSON.stringify(payload)
  });
}

function enviarReporteLumiere() {
  const cfg = {
    to: "equipo@tuagencia.com",
    subject: `✦ Reporte IA · ${hoy()}`,
    api: PropertiesService
      .getScriptProperties()
      .getProperty("LUMIERE_API_KEY")
  };

  // 1. Obtener reporte desde Lumière API
  const res = UrlFetchApp.fetch(
    "https://api.lumiere.ai/v1/report", {
    method: "POST",
    headers: {
      "Authorization": `Bearer ${cfg.api}`,
      "Content-Type": "application/json"
    },
    payload: JSON.stringify({ date: hoy() })
  });

  const data = JSON.parse(res.getContentText());

  // 2. Enviar por Gmail
  GmailApp.sendEmail(cfg.to, cfg.subject,
    "", {
      htmlBody: buildHTML(data),
      name: "Neuro AI"
    }
  );
}

function hoy() {
  return Utilities.formatDate(
    new Date(), "America/Bogota",
    "dd/MM/yyyy"
  );
}
